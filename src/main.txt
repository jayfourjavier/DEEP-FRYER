#include <Arduino.h>
#include <Adafruit_MAX31865.h>
#include "Relay.h"
#include "Buzzer.h"

Adafruit_MAX31865 pt100 = Adafruit_MAX31865(5);

#define HEATER_RELAY_PIN 33
#define UPWARD_RELAY_PIN 26
#define DOWNWARD_RELAY_PIN 25
#define BUZZER_PIN 4

// The value of the Rref resistor. Use 430.0 for PT100 and 4300.0 for PT1000
#define RREF 430.0
#define RNOMINAL 100.0

Relay heaterRelay(HEATER_RELAY_PIN, "Heater Relay", true);
Relay upwardRelay(UPWARD_RELAY_PIN, "Upward Relay", true);
Relay downwardRelay(DOWNWARD_RELAY_PIN, "Downward Relay", true);
Buzzer buzzer(BUZZER_PIN, "Buzzer", true);

void testPt100()
{
  uint16_t rtd = pt100.readRTD();

  Serial.print("RTD value: ");
  Serial.println(rtd);
  float ratio = rtd;
  ratio /= 32768;
  Serial.print("Ratio = ");
  Serial.println(ratio, 8);
  Serial.print("Resistance = ");
  Serial.println(RREF * ratio, 8);
  Serial.print("Temperature = ");
  Serial.println(pt100.temperature(RNOMINAL, RREF));

  // Check and print any faults
  uint8_t fault = pt100.readFault();
  if (fault)
  {
    Serial.print("Fault 0x");
    Serial.println(fault, HEX);
    if (fault & MAX31865_FAULT_HIGHTHRESH)
    {
      Serial.println("RTD High Threshold");
    }
    if (fault & MAX31865_FAULT_LOWTHRESH)
    {
      Serial.println("RTD Low Threshold");
    }
    if (fault & MAX31865_FAULT_REFINLOW)
    {
      Serial.println("REFIN- > 0.85 x Bias");
    }
    if (fault & MAX31865_FAULT_REFINHIGH)
    {
      Serial.println("REFIN- < 0.85 x Bias - FORCE- open");
    }
    if (fault & MAX31865_FAULT_RTDINLOW)
    {
      Serial.println("RTDIN- < 0.85 x Bias - FORCE- open");
    }
    if (fault & MAX31865_FAULT_OVUV)
    {
      Serial.println("Under/Over voltage");
    }
    pt100.clearFault();
  }
  Serial.println();
  delay(1000);
}

void setup()
{
  Serial.begin(115200);
  Serial.println("Adafruit MAX31865 PT100 Sensor Test!");

  pt100.begin(MAX31865_3WIRE); // set to 2WIRE or 4WIRE as necessary

  heaterRelay.begin();
  upwardRelay.begin();
  downwardRelay.begin();
  buzzer.begin();
  buzzer.mode(BUZZER_READY);
}

  static int testMode = BUZZER_BEEP;
  static unsigned long lastTime = 0;

  // Wait until current buzzer pattern finishes
  if (buzzer.isPlaying())
    return;

  // 1 second pause between patterns
  if (millis() - lastTime < 1000)
    return;

  lastTime = millis();

  switch (testMode)
  {
  case BUZZER_BEEP:
    Serial.println("[BUZZER] BEEP");
    buzzer.mode(BUZZER_BEEP);
    testMode = BUZZER_READY;
    break;

  case BUZZER_READY:
    Serial.println("[BUZZER] READY");
    buzzer.mode(BUZZER_READY);
    testMode = BUZZER_START;
    break;

  case BUZZER_START:
    Serial.println("[BUZZER] START");
    buzzer.mode(BUZZER_START);
    testMode = BUZZER_LOWERING;
    break;

  case BUZZER_LOWERING:
    Serial.println("[BUZZER] LOWERING");
    buzzer.mode(BUZZER_LOWERING);
    testMode = BUZZER_RAISING;
    break;

  case BUZZER_RAISING:
    Serial.println("[BUZZER] RAISING");
    buzzer.mode(BUZZER_RAISING);
    testMode = BUZZER_DONE;
    break;

  case BUZZER_DONE:
    Serial.println("[BUZZER] DONE");
    buzzer.mode(BUZZER_DONE);
    testMode = BUZZER_ERROR;
    break;

  case BUZZER_ERROR:
    Serial.println("[BUZZER] ERROR");
    buzzer.mode(BUZZER_ERROR);
    testMode = BUZZER_ALARM;
    break;

  case BUZZER_ALARM:
    Serial.println("[BUZZER] ALARM");
    buzzer.mode(BUZZER_ALARM);
    testMode = BUZZER_BEEP;
    break;
  }
}
